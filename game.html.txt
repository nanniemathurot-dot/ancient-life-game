    // Audio Synthesizer (Web Audio API)
    let audioCtx = null;
    let audioEnabled = true;

    function initAudio() {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }

    function playTone(freq, type, duration) {
        if (!audioEnabled) return;
        initAudio();
        try {
            let osc = audioCtx.createOscillator();
            let gain = audioCtx.createGain();
            osc.type = type;
            osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
            gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + duration);
        } catch(e) {}
    }

    function toggleAudio() {
        audioEnabled = !audioEnabled;
        document.querySelector('.audio-btn').innerText = audioEnabled ? '🔊 เสียง' : '🔇 ปิดเสียง';
    }

    function triggerRandomEvent() {
        let ev = events[Math.floor(Math.random() * events.length)];
        document.getElementById('modal-title').innerText = ev.title;
        document.getElementById('modal-desc').innerText = ev.desc;
        let choicesDiv = document.getElementById('modal-choices');
        choicesDiv.innerHTML = "";
        
        ev.choices.forEach(ch => {
            let btn = document.createElement('button');
            btn.className = 'choice-btn';
            btn.innerText = ch.text;
            btn.onclick = () => {
                ch.effect();
                document.getElementById('event-modal').style.display = 'none';
                updateUI();
            };
            choicesDiv.appendChild(btn);
        });
        document.getElementById('event-modal').style.display = 'flex';
    }

    function takeExam() {
        if(player.age < 16) {
            alert("อายุยังน้อยเกินไป ต้องมีอายุอย่างน้อย 16 ปีจึงจะสามารถสอบขุนนางได้!");
            return;
        }
        playTone(783.99, 'sine', 0.4);
        if(player.intel >= 65) {
            player.rank = "ขุนนางขั้น 9 (จอหงวน)";
            player.rankLevel = 1;
            logEvent("สอบผ่านจอหงวนในราชสำนัก ได้รับการแต่งตั้งเป็นขุนนางขั้น 9!");
            alert("ยินดีด้วย! สอบผ่านเป็นขุนนางราชสำนักสำเร็จ!");
        } else {
            logEvent("สอบตกในการคัดเลือกขุนนาง ความรู้ยังไม่เพียงพอ");
            alert("น่าเสียดาย... สอบตก! สติปัญญาของคุณยังต้องพัฒนาเพิ่มขึ้น");
        }
        updateUI();
    }

    function doWork() {
        if(player.rankLevel === 0) {
            alert("คุณยังไม่ได้รับราชการ โปรดสอบขุนนางให้ผ่านก่อน!");
            return;
        }
        playTone(523.25, 'sine', 0.2);
        let salary = player.rankLevel * 80;
        player.money += salary;
        logEvent(`ปฏิบัติราชการในหน้าที่ ได้รับเบี้ยหวัด ${salary} ตำลึง`);
        updateUI();
    }
